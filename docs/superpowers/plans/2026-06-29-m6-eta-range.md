# M-Track-6: ETA Range (lower/nominal/upper) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Widen the single CI ETA into an optimistic/nominal/pessimistic range (p10/p50/p90), rendered as a Plotroom-style confidence band (solid elapsed → faded p10→p90 → dark p50 tick) on both the PR CI stage connector and each running drill-down row.

**Architecture:** Extract the estimator's per-check remaining-time logic into one pure helper, `check_estimate`, returning a 3-bound `CheckEstimate`. `compute_progress` aggregates it (max per bound) for the PR; `build_check_groups` calls it per running row. Three numbers thread through the wire (`PullRequestInfo.eta_low/high_seconds`, `CheckRow.rem_{low,nominal,high}_seconds`); one shared frontend geometry helper (`bandSegments`) drives both render sites.

**Tech Stack:** Python 3.11 / dataclasses / pytest (backend); React + TypeScript + Vite / vitest (frontend).

## Global Constraints

- **Spec:** `docs/superpowers/specs/2026-06-29-visual-pipeline-track-m6-eta-range-design.md`.
- **Bounds:** lower = `Expected.p10`, nominal = `Expected.p50`, upper = `Expected.p90` (all already stored). No new GitHub data, no new query, no new percentiles.
- **One shared spine:** the per-check remaining computation lives ONLY in `check_estimate`; the estimator and the drill-down both call it (no parallel implementation — constitution VI).
- **Nominal unchanged:** `ProgressResult.eta_seconds` / `eta_percent` / `overdue` keep their exact current meaning and values on existing fixtures. `eta_low_seconds`/`eta_high_seconds` are additive.
- **overdue → no range:** when the PR is overdue, all three ETA fields are `None` (preserve today's short-circuit). The per-row overdue rule in `checkgroups._row` is UNCHANGED (not replaced by `check_estimate.overdue`).
- **Store-tolerant rows / store-gated PR ETA (the #351 invariant):** drill-down rows still build when `store is None` — they just carry `rem_* = None` (no band). PR-level ETA still only computes when a store is present.
- **Ordering holds:** `eta_low ≤ eta_seconds ≤ eta_high` (max-aggregation preserves per-check `p10≤p50≤p90`).
- **Timezone:** `now` is injected and tz-aware; parse ISO with `Z→+00:00`. Never naive `utcnow` (memory: `naive-utc-isoformat-0s-ago`).

```python
# server/pipeline/check_eta.py
"""Pure per-check estimate: weight, progress, and a p10/p50/p90 remaining range.

The single source for "remaining time for one check" — consumed by the PR-level
estimator (aggregated) and the per-row drill-down (per running row). Ported from
estimator.compute_progress's inner loop, generalised from p50/p90 to p10/p50/p90.
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime

from server.pipeline.history import Expected, median, percentile

RUNNING_CAP = 0.97
OVERDUE_FACTOR = 1.5
DEFAULT_DURATION = 60.0


@dataclass(frozen=True)
class CheckEstimate:
    weight: float          # p50 used for percent weighting
    progress: float        # 0..1
    rem_low: float         # seconds remaining, optimistic (p10)
    rem_nominal: float     # seconds remaining, nominal (p50)
    rem_high: float        # seconds remaining, pessimistic (p90)
    overdue: bool
    unfinished: bool


def _parse_epoch_s(iso: str | None) -> float:
    if not iso:
        return float("nan")
    try:
        return datetime.fromisoformat(iso.replace("Z", "+00:00")).timestamp()
    except ValueError:
        return float("nan")


def check_estimate(
    check: dict | None,
    expected: Expected | None,
    samples: list[float],
    now: datetime,
    fallback: float,
    queue_delay: float | None = None,
) -> CheckEstimate | None:
    c = check
    if c is not None and c.get("conclusion") == "SKIPPED":
        return None

    p10 = expected.p10 if expected else fallback
    p50 = expected.p50 if expected else fallback
    p90 = expected.p90 if expected else p50
    weight = p50
    status = c.get("status") if c else None
    overdue = False
    now_s = now.timestamp()

    if status == "COMPLETED":
        return CheckEstimate(weight, 1.0, 0.0, 0.0, 0.0, False, False)

    if status == "IN_PROGRESS" and c and c.get("started_at"):
        raw_elapsed = now_s - _parse_epoch_s(c["started_at"])
        if raw_elapsed != raw_elapsed:  # NaN
            return CheckEstimate(weight, 0.0, p10, p50, p90, False, True)
        elapsed = max(raw_elapsed, 0.0)
        progress = min(elapsed / p50, RUNNING_CAP) if p50 else RUNNING_CAP
        vals = samples or []
        qualifying = [v for v in vals if v > elapsed] if len(vals) >= 5 else None
        if qualifying is not None and len(qualifying) == 0:
            if len(vals) >= 10:
                overdue = True
            elif elapsed > OVERDUE_FACTOR * p50:
                overdue = True
            rem_low = rem_nom = rem_high = 0.0
        elif qualifying is not None and len(qualifying) >= 2:
            sq = sorted(qualifying)
            rem_low = max(percentile(sq, 0.1) - elapsed, 0.0)
            rem_nom = median(sq) - elapsed
            rem_high = percentile(sq, 0.9) - elapsed
            denom = elapsed + rem_nom
            progress = min(elapsed / denom, RUNNING_CAP) if denom else RUNNING_CAP
        else:
            if elapsed > OVERDUE_FACTOR * p50:
                overdue = True
            rem_low = max(p10 - elapsed, 0.0)
            rem_nom = max(p50 - elapsed, 0.0)
            rem_high = max(p90 - elapsed, 0.0)
        return CheckEstimate(weight, progress, rem_low, rem_nom, rem_high, overdue, True)

    # queued, or expected-from-history but not yet in the rollup
    delay = (queue_delay or 0.0)
    return CheckEstimate(weight, 0.0, p10 + delay, p50 + delay, p90 + delay, False, True)
```

// web/src/lib/etaBand.ts
export interface BandInput { elapsedS: number; lowS: number; nominalS: number; highS: number }
export interface BandSegments { solidPct: number; bandLeftPct: number; bandWidthPct: number; tickPct: number }

const clamp01 = (x: number) => Math.max(0, Math.min(1, x))

export function bandSegments(i: BandInput): BandSegments | null {
  const { elapsedS, lowS, nominalS, highS } = i
  for (const v of [elapsedS, lowS, nominalS, highS]) {
    if (v == null || Number.isNaN(v)) return null
  }
  const scale = elapsedS + highS
  if (scale <= 0) return null
  const bandLeft = clamp01((elapsedS + lowS) / scale)
  return {
    solidPct: clamp01(elapsedS / scale) * 100,
    bandLeftPct: bandLeft * 100,
    bandWidthPct: (1 - bandLeft) * 100,         // band spans low-completion → scale (= high completion)
    tickPct: clamp01((elapsedS + nominalS) / scale) * 100,
  }
}

// Node adapter: the PR aggregate has no single elapsed; derive one from percent + nominal.
export function pseudoElapsedFromPercent(nominalS: number, percent: number): number {
  return (nominalS * percent) / Math.max(100 - percent, 1)
}
